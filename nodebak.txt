logging.basicConfig(level=logging.DEBUG)

SOFT_VOTE_TIMEOUT = 0.6
DEFER_TIMEOUT = 1.2
VIEW_TIMEOUT = 3.0

class Node:
    def __init__(self, node_id, priv_key, pub_key, network, f=1, all_nodes=None):
        self.id = node_id
        self.priv = priv_key   # for mock: just node id string used in crypto.sign
        self.pub = pub_key
        self.network = network
        self.state = NodeState(f=f)
        self.view = 0
        self.is_leader = False
        self.all_nodes = all_nodes or []

        # inside __init__ DID
        self.world_state = WorldState()
        self.metrics = None  # 在 simulator 启动时由外部注入 shared Metrics 实例
        self.did_pub_lookup = {}  # 由 simulator 注入： did -> public_key

        # storage for partial signatures: {block_id: [msg,...]}
        self.partial_sigs = {}
        # partial_qcs store
        self.partial_qcs = {}

        # defer notices: {view: set(node_ids)}
        self.defer_notices = {}

        # timer placeholders
        self._soft_timer = None
        self._defer_timer = None
        self._view_timer = None

    @staticmethod
    def tx_hash(tx: dict) -> str:
        """
        计算交易的哈希值
        :param tx: 交易字典
        :return: 交易的哈希值
        """
        s = json.dumps({k: v for k, v in tx.items() if k != "signature"}, sort_keys=True, separators=(",", ":"))
        return hashlib.sha256(s.encode()).hexdigest()

    # -------------------------------------------------------
    # receive_message: network will call this
    # -------------------------------------------------------
    def receive_message(self, msg_type, payload):
        if msg_type == "proposal":
            self.on_receive_proposal(payload)
        elif msg_type == "soft_vote":
            self.on_receive_soft_vote(payload)
        elif msg_type == "partial_signature":
            self.receive_partial_signature(payload)
        elif msg_type == "partial_qc":
            self.receive_partial_qc(payload)
        elif msg_type == "defer":
            self.on_receive_defer(payload)
        elif msg_type == "vote":
            self.on_receive_vote(payload)
        elif msg_type == "view_change":
            self.receive_view_change(payload)
        else:
            logging.warning(f"[{self.id}] Unknown msg_type {msg_type}")

    # -------------------------------------------------------
    # proposal handling
    # -------------------------------------------------------
    def on_receive_proposal(self, proposal):
        logging.debug(f"[{self.id}] Received proposal: {proposal}")
        # 检查提案是否合法
        if not self.validate_proposal(proposal):
            logging.warning(f"[{self.id}] Invalid proposal: {proposal}")
            return
        # proposal is a dict: {block: Block, qc: QC or None, view: v}
        block = proposal["block"]
        qc = proposal.get("qc")
        view = proposal.get("view", 0)
        txs = proposal.get("transactions", [])
        logging.debug(f"[{self.id}] Got proposal {block} view={view}")

        from transaction import verify_tx_signature

        for tx in txs:
            ok = verify_tx_signature(tx, self.did_pub_lookup)
            if not ok:
                logging.warning(f"[{self.id}] Proposal contains invalid tx signature: {tx}")
                # reject tx or reject proposal; here我们把 proposal 当作 invalid -> soft-vote
                self.soft_vote(block, view)
                return
            
        if self.metrics:
            for tx in txs:
                txid = Node.tx_hash(tx)  # 请参见下面的 tx_hash helper
                self.metrics.mark_included(txid)
        # store block & txs to local state so commit can apply them later
        self.state.add_block(block)
        block.payload = {"transactions": txs}

        # validate block basic
        if not block.validate():
            logging.warning(f"[{self.id}] invalid block {block}")
            return

        # check QC conflict
        if qc and is_conflicting_qc(qc, getattr(self.state, "latest_qc", None)):
            logging.warning(f"[{self.id}] detected conflicting QC, entering soft-vote for {block.id}")
            self.soft_vote(block, view)
            return

        # normal path: vote
        self.vote(block, view)

    def vote(self, block, view):
        logging.debug(f"[{self.id}] voting for {block.id} (v{view})")
        # 确保 block.id 是字节类型
        sig = BLS.sign(self.priv, block.id.encode())
        vote = Vote(self.id, block.id, sig, view)
        self.network.broadcast_vote(self.id, vote.to_dict())
        self.state.record_vote(block.id)
        # broadcast partial sig so leader can build partial QC
        self.broadcast_partial_signature(block, view)

    def soft_vote(self, block, view):
        logging.debug(f"[{self.id}] soft-vote for {block.id} (v{view})")
        self.network.broadcast_soft_vote(self.id, block.id)
        # still broadcast partial sig as evidence
        self.broadcast_partial_signature(block, view)
        # start soft timeout -> may later send defer
        self.start_soft_timer(view)

    def broadcast_partial_signature(self, block, view):
        # sig = sign(self.id, f"partial:{block.id}:{view}")
        sig = BLS.sign(self.priv, f"partial:{block.id}:{view}")
        msg = {
            "block_id": block.id,
            "view": view,
            "high_qc_view": getattr(self.state.latest_qc, "view", 0),
            "signature": sig,
            "signer": self.id
        }
        self.network.broadcast_partial_signature(self.id, msg)

    # -------------------------------------------------------
    # partial signature handling (leader collects)
    # -------------------------------------------------------
    def receive_partial_signature(self, msg):
        bid = msg["block_id"]
        self.partial_sigs.setdefault(bid, [])
        # avoid duplicates by signer
        if any(p["signer"] == msg["signer"] for p in self.partial_sigs[bid]):
            return
        self.partial_sigs[bid].append(msg)
        logging.debug(f"[{self.id}] received partial sig for {bid} from {msg['signer']} ({len(self.partial_sigs[bid])})")

        # if leader, try form partial QC or full QC
        if self.is_leader:
            # if >= f+1 form partial QC
            if len(self.partial_sigs[bid]) >= self.state.f_plus_1 and bid not in self.partial_qcs:
                partials = [p["signature"] for p in self.partial_sigs[bid]]
                agg = aggregate(partials)
                pqc = PartialQC(block_id=bid, view=msg["view"], aggregated_sig=agg, signer_count=len(partials))
                self.partial_qcs[bid] = pqc
                logging.info(f"[{self.id}] Leader formed PartialQC for {bid} (t={pqc.signer_count})")
                # broadcast partial QC
                self.network.broadcast_partial_qc(self.id, pqc.to_dict())

            # if >= 2f+1 form full QC (in mock we reuse aggregate)
            if len(self.partial_sigs[bid]) >= self.state.full_q_threshold:
                partials = [p["signature"] for p in self.partial_sigs[bid]]
                agg_all = aggregate(partials)
                signers = [p["signer"] for p in self.partial_sigs[bid]]
                qc = QC(block_id=bid, signatures=partials, signers=signers, view=msg["view"], agg=agg_all)
                logging.info(f"[{self.id}] Leader formed FULL QC for {bid} (>=2f+1). Broadcasting to nodes.")
                # leader broadcasts qc to all (as proposal's qc or directly)
                self.broadcast_qc(qc)

    def receive_partial_qc(self, msg):
        # store partial QC announced by some leader
        bid = msg["block_id"]
        pqc = PartialQC(block_id=bid, view=msg["view"], aggregated_sig=msg["aggregated_sig"], signer_count=msg["signer_count"])
        self.partial_qcs[bid] = pqc
        logging.debug(f"[{self.id}] stored PartialQC for {bid} (t={pqc.signer_count})")

    def broadcast_qc(self, qc):
        # send as a special message (here we reuse 'proposal' to distribute qc knowledge)
        data = {"block": self.state.get_block(qc.block_id), "qc": qc, "view": qc.view}
        self.network.broadcast(self.id, "qc_announce", data)

    # -------------------------------------------------------
    # defer handling (soft timeout)
    # -------------------------------------------------------
    def start_soft_timer(self, view):
        if self._soft_timer:
            self._soft_timer.cancel()
        def on_timeout():
            # send defer notice
            logging.debug(f"[{self.id}] soft timeout expired for v{view}, broadcasting defer")
            self.send_defer(view)
            # start defer timer waiting for 2f+1 defers or view-change
            self.start_defer_timer(view)
        self._soft_timer = threading.Timer(SOFT_VOTE_TIMEOUT, on_timeout)
        self._soft_timer.start()

    def send_defer(self, view):
        msg = {"sender": self.id, "view": view}
        self.network.broadcast(self.id, "defer", msg)
        # also track locally
        self.defer_notices.setdefault(view, set()).add(self.id)

    def on_receive_defer(self, msg):
        view = msg["view"]
        sender = msg["sender"]
        self.defer_notices.setdefault(view, set()).add(sender)
        cnt = len(self.defer_notices[view])
        logging.debug(f"[{self.id}] received defer from {sender} for v{view} ({cnt})")
        if cnt >= self.state.full_q_threshold:
            logging.warning(f"[{self.id}] collected >=2f+1 defers for v{view} -> trigger view change")
            self.handle_view_change(view+1)

    def start_defer_timer(self, view):
        if self._defer_timer:
            self._defer_timer.cancel()
        def on_defertout():
            logging.debug(f"[{self.id}] defer timer expired for v{view}, attempting view change")
            self.handle_view_change(view+1)
        self._defer_timer = threading.Timer(DEFER_TIMEOUT, on_defertout)
        self._defer_timer.start()

    # -------------------------------------------------------
    # view change
    # -------------------------------------------------------
    def handle_view_change(self, new_view):
        logging.info(f"[{self.id}] handling view change to v{new_view}")
        self.view = new_view
        # if I am the new leader, collect partial evidence (in this demo leader selection done external)
        # broadcast view-change message containing local partial_qcs and latest_qc
        leader_id = self.all_nodes[new_view % len(self.all_nodes)]
        self.is_leader = (self.id == leader_id)
        logging.info(f"[{self.id}] New leader for view {new_view} is {leader_id}")
        vc = {
            "new_view": new_view,
            "highest_qc": getattr(self.state.latest_qc, "to_dict", lambda: None)(),
            "partial_qcs": {k: v.to_dict() for k,v in self.partial_qcs.items()},
            "locked_block": getattr(self.state.locked_qc, "block_id", None),
            "sender": self.id
        }
        self.network.broadcast_view_change(self.id, vc)

    def receive_view_change(self, msg):
        # This method is invoked when nodes receive other's view-change
        logging.debug(f"[{self.id}] receive_view_change from {msg.get('sender')}, new_view={msg.get('new_view')}")

    # -------------------------------------------------------
    # receiving normal votes (not used for mock full QC)
    # -------------------------------------------------------
    def on_receive_vote(self, vote_dict):
        logging.debug(f"[{self.id}] Received vote: {vote_dict}")
        # 检查 vote_dict 是否包含必要字段
        # if "signer" not in vote_dict or "block_id" not in vote_dict:
        #     logging.error(f"[{self.id}] Received invalid vote_dict: {vote_dict}")
        #     return
        block_id = vote_dict["block_id"]
        # self.partial_sigs.setdefault(block_id, [])
        # if any(v["signer"] == vote_dict["signer"] for v in self.partial_sigs[block_id]):
        #     return
        # self.partial_sigs[block_id].append(vote_dict)
        # logging.debug(f"[{self.id}] Collected {len(self.partial_sigs[block_id])} votes for block {block_id}")

        # 如果是领导者，尝试形成 QC
        if self.is_leader:
            if len(self.partial_sigs[block_id]) >= self.state.f_plus_1:
                partials = [v["signature"] for v in self.partial_sigs[block_id]]
                agg = aggregate(partials)
                qc = QC(block_id=block_id, signatures=partials, signers=[v["signer"] for v in self.partial_sigs[block_id]], view=vote_dict["view"], agg=agg)
                logging.info(f"[{self.id}] Formed QC for block {block_id}")
                self.broadcast_qc(qc)

    # -------------------------------------------------------
    # qc announcement handler (leader broadcast of a full qc)
    # -------------------------------------------------------
    def receive_message_qc_announce(self, data):
        block = data.get("block")
        qc = data.get("qc")
        logging.info(f"[{self.id}] Received QC announce for block {qc.block_id}")
        # update latest qc and locked qc if applicable
        self.state.update_latest_qc(qc)
        self.state.update_locked_qc(qc)
        # maybe commit if view advanced
        if qc.view > getattr(self.state.commit_qc, "view", 0):
            logging.info(f"[{self.id}] Committing block {qc.block_id}")
            self.commit_block(qc.block_id)

    # unify qc_announce handler into receive_message dispatcher
    def receive_message(self, msg_type, payload):
        # override to route qc_announce correctly
        if msg_type == "qc_announce":
            self.receive_message_qc_announce(payload)
            return
        # otherwise previous behavior
        if msg_type == "proposal":
            self.on_receive_proposal(payload)
        elif msg_type == "soft_vote":
            self.on_receive_soft_vote(payload)
        elif msg_type == "partial_signature":
            self.receive_partial_signature(payload)
        elif msg_type == "partial_qc":
            self.receive_partial_qc(payload)
        elif msg_type == "defer":
            self.on_receive_defer(payload)
        elif msg_type == "vote":
            self.on_receive_vote(payload)
        elif msg_type == "view_change":
            self.receive_view_change(payload)
        else:
            logging.warning(f"[{self.id}] Unknown msg_type {msg_type}")

    # small helper for soft_vote arrival (not used heavily)
    def on_receive_soft_vote(self, payload):
        # for demo we just log
        logging.debug(f"[{self.id}] soft_vote received payload={payload}")

    # -------------------------------------------------------
    # commit
    # -------------------------------------------------------
    def commit_block(self, block_id):
        logging.info(f"[{self.id}] COMMITTED block {block_id}")
        blk = self.state.get_block(block_id)
        if blk and getattr(blk, "payload", None):
            txs = blk.payload.get("transactions", [])
            for tx in txs:
                applied = self.world_state.apply_transaction(tx)
                # metrics
                if self.metrics:
                    txid = Node.tx_hash(tx)
                    self.metrics.mark_committed(txid)
        # update commit_qc pointer
        self.state.commit_qc = self.state.latest_qc

    def validate_proposal(self, proposal):
            """
            验证提案的合法性
            """
            block = proposal.get("block")
            qc = proposal.get("qc")
            view = proposal.get("view", 0)

            if not block:
                logging.warning(f"[{self.id}] Proposal missing block")
                return False

            # if block.parent_id and not self.state.get_block(block.parent_id):
            #     logging.warning(f"[{self.id}] Proposal references unknown parent block {block.parent_id}")
            #     return False

            if view < self.view:
                logging.warning(f"[{self.id}] Proposal view {view} is stale (current view: {self.view})")
                return False

            if qc and not self.validate_qc(qc):
                logging.warning(f"[{self.id}] Proposal contains invalid QC")
                return False

            return True

    def validate_qc(self, qc):
        """
        验证 QC 的合法性
        """
        # 如果是 DummyQC，直接跳过验证
        if isinstance(qc, DummyQC):
            logging.debug(f"[{self.id}] Skipping validation for DummyQC")
            return True
        
        if not verify_aggregate(qc.signers, qc.block_id.encode(), qc.agg):
            logging.warning(f"[{self.id}] QC signature verification failed for block {qc.block_id}")
            return False

        if qc.view < getattr(self.state.latest_qc, "view", 0):
            logging.warning(f"[{self.id}] QC view {qc.view} is stale")
            return False

        return True
